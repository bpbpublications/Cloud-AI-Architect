Section 6.6.2  Building text classification with Gemini, replacing AutoML NLP
Code block 6.6.2

from google import genai
from google.genai import types

client = genai.Client(vertexai=True,
                      project="your-project-id",
                      location="us-central1")

LABELS = ["billing", "outage", "how-to", "feature-request"]

# Model IDs retire on a published schedule; check the current
# model versions page before pinning one in production code.
MODEL = "gemini-2.5-flash"

def classify(ticket_text):
    response = client.models.generate_content(
        model=MODEL,
        contents=ticket_text,
        config=types.GenerateContentConfig(
            system_instruction=(
                "Classify the support ticket into exactly one of: "
                + ", ".join(LABELS)
                + ". Reply with the label only."
            ),
            temperature=0,
        ),
    )
    return response.text.strip()

Section 6.7  Model discovery and reuse with Vertex AI Model Registry and Model Garden
Code block 6.7
------------------------------------------------------------------------------
from kfp import compiler, dsl
from google_cloud_pipeline_components.v1.dataset import (
    ImageDatasetCreateOp)
from google_cloud_pipeline_components.v1.automl.training_job import (
    AutoMLImageTrainingJobRunOp)
from google_cloud_pipeline_components.v1.endpoint import (
    EndpointCreateOp, ModelDeployOp)

@dsl.pipeline(
    name="automl-vision-retrain",
    pipeline_root="gs://your-bucket/pipeline-root",
)
def retrain_pipeline(project: str, location: str = "us-central1"):
    dataset = ImageDatasetCreateOp(
        project=project,
        location=location,
        display_name="product-classification-dataset",
        gcs_source="gs://your-bucket/import_file.csv",
        import_schema_uri="gs://google-cloud-aiplatform/schema/dataset/"
                          "ioformat/image_classification_single_label_io_"
                          "format_1.0.0.yaml",
    )
    training = AutoMLImageTrainingJobRunOp(
        project=project,
        location=location,
        display_name="product-classification-training",
        prediction_type="classification",
        model_type="CLOUD",
        dataset=dataset.outputs["dataset"],
        budget_milli_node_hours=8000,
    )
    endpoint = EndpointCreateOp(
        project=project,
        location=location,
        display_name="product-classifier-endpoint",
    )
    ModelDeployOp(
        model=training.outputs["model"],
        endpoint=endpoint.outputs["endpoint"],
        dedicated_resources_machine_type="n1-standard-4",
        dedicated_resources_min_replica_count=1,
        dedicated_resources_max_replica_count=5,
    )

compiler.Compiler().compile(retrain_pipeline, "pipeline.yaml")

Code block 6.8.1

endpoint = aiplatform.Endpoint.create(     display_name="product-classifier-endpoint",     description="AutoML image classification for product catalog", )  model.deploy(     endpoint=endpoint,     deployed_model_display_name="product-classifier-v1-deployed",     machine_type="n1-standard-4",     min_replica_count=1,     max_replica_count=5,     accelerator_type=None,     accelerator_count=0,     traffic_split={"0": 100}, )
In the traffic_split mapping, the "0" key is easy to misread as a model ID or a version number. It is neither: it is a placeholder for the model being deployed in this call, and Vertex AI substitutes the deployed model's real ID once deployment completes. Splits across two live versions use the actual deployed model IDs as keys.

Section 6.8.2  Batch prediction configuration
Code block 6.8.2
------------------------------------------------------------------------------
batch_prediction_job = model.batch_predict(     job_display_name="churn-batch-prediction-weekly",     instances_format="bigquery",     predictions_format="bigquery",     bigquery_source="bq://your-project.your-dataset.customers_to_score",     bigquery_destination_prefix="bq://your-project.your-dataset.churn_predictions",     machine_type="n1-standard-4",     starting_replica_count=5,     max_replica_count=10, )  batch_prediction_job.wait() print(f"Batch prediction state: {batch_prediction_job.state}")
Size starting_replica_count and max_replica_count against the job rather than copying these values. The starting count sets the parallelism the job begins with and the maximum caps it, so larger inputs and tighter completion windows justify higher counts, while a small recurring job finishes just as quickly on one or two replicas and costs less. Because the cluster is torn down at the end of the job, over-provisioning costs you throughput you did not need rather than idle capacity you keep paying for.

Section 6.10  SRE best practices for production AutoML pipelines
Code block 6.10
------------------------------------------------------------------------------
from google.cloud import aiplatform
from google.cloud.aiplatform import model_monitoring

aiplatform.init(project="your-project-id", location="us-central1")

# Compare live traffic against the distribution seen in training.
skew = model_monitoring.SkewDetectionConfig(
    data_source="bq://your-project.your-dataset.churn_training_table",
    target_field="churned",
    skew_thresholds={"account_age_days": 0.3, "plan_type": 0.3},
)

# Compare recent traffic against earlier traffic.
drift = model_monitoring.DriftDetectionConfig(
    drift_thresholds={"account_age_days": 0.3, "plan_type": 0.3},
)

monitoring_job = aiplatform.ModelDeploymentMonitoringJob.create(
    display_name="churn-endpoint-monitoring",
    endpoint=endpoint,
    logging_sampling_strategy=model_monitoring.RandomSampleConfig(
        sample_rate=0.2),
    schedule_config=model_monitoring.ScheduleConfig(monitor_interval=1),
    alert_config=model_monitoring.EmailAlertConfig(
        user_emails=["ml-oncall@your-company.com"],
        enable_logging=True),
    objective_configs=model_monitoring.ObjectiveConfig(skew, drift),
)
