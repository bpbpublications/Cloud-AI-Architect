    credential=credential,
    subscription_id="<subscription-id>",
    resource_group_name="<resource-group>",
    workspace_name="<workspace-name>"
)

    type=AssetTypes.MLTABLE,
    path="azureml:customer-churn-train:1"

    compute="cpu-cluster-4x",
    experiment_name="churn-automl-experiment",
    training_data=training_data,
    target_column_name="churn_flag",
    primary_metric="AUC_weighted",
    n_cross_validations=5,
    enable_model_explainability=True,
    tags={"project": "churn-prediction", "environment": "dev"}
)

    timeout_minutes=120,
    trial_timeout_minutes=20,
    max_trials=30,
    max_concurrent_trials=6
)

from azure.ai.ml import MLClient
from azure.ai.ml.entities import (
    ManagedOnlineEndpoint, ManagedOnlineDeployment, CodeConfiguration
)
from azure.identity import DefaultAzureCredential

credential = DefaultAzureCredential()
ml_client = MLClient(
    credential=credential,
    subscription_id="<subscription-id>",
    resource_group_name="<resource-group>",
    workspace_name="<workspace-name>"
)

# Create the endpoint resource
endpoint = ManagedOnlineEndpoint(
    name="churn-prediction-endpoint",
    description="Production churn prediction endpoint",
    auth_mode="key",
    tags={"project": "churn", "environment": "prod"}
)
ml_client.online_endpoints.begin_create_or_update(endpoint).result()

# Create the deployment under the endpoint
deployment = ManagedOnlineDeployment(
    name="churn-v1",
    endpoint_name="churn-prediction-endpoint",
    model="azureml:churn-model:3",
    code_configuration=CodeConfiguration(
        code="./inference",
        scoring_script="score.py"
    ),
    environment="azureml:sklearn-inference:1",
    instance_type="Standard_DS3_v2",
    instance_count=2
)
ml_client.online_deployments.begin_create_or_update(deployment).result()

# Re-fetch first so concurrent changes are not clobbered, then shift traffic
endpoint = ml_client.online_endpoints.get("churn-prediction-endpoint")
ml_client.online_endpoints.begin_create_or_update(endpoint).result()
print("Model deployed and traffic routed.")  # canary: {"churn-v1": 90, "churn-v2": 10}

# score.py
import json, os
import joblib
import numpy as np

model = None

def init():
    global model
    model_path = os.path.join(os.environ["AZUREML_MODEL_DIR"], "model.pkl")
    model = joblib.load(model_path)

def run(raw_data):
    try:
        data = json.loads(raw_data)
        features = np.array(data["features"])
        predictions = model.predict(features)
        return {"predictions": predictions.tolist()}
    except Exception as e:
        return {"error": str(e)}

# rag_agent.py
from langchain_openai import AzureChatOpenAI  # langchain-openai 0.2.x
from langchain_openai import AzureOpenAIEmbeddings
from langchain_community.vectorstores import AzureSearch
from langchain.chains import create_retrieval_chain
from langchain.prompts import PromptTemplate
from azure.identity import DefaultAzureCredential, get_bearer_token_provider

import os

token_provider = get_bearer_token_provider(
    DefaultAzureCredential(), "https://cognitiveservices.azure.com/.default"
)

# Initialize Azure OpenAI LLM endpoint
llm = AzureChatOpenAI(
    azure_deployment=os.environ["AZURE_OPENAI_DEPLOYMENT"],
    azure_endpoint=os.environ["AZURE_OPENAI_ENDPOINT"],
    azure_ad_token_provider=token_provider,
    api_version="2024-02-01",
    temperature=0.0,
    max_tokens=1024
)

# Initialize embedding model for document retrieval
embeddings = AzureOpenAIEmbeddings(
    azure_deployment="text-embedding-ada-002",
    azure_endpoint=os.environ["AZURE_OPENAI_ENDPOINT"],
    azure_ad_token_provider=token_provider
)

# Connect to Azure Cognitive Search vector store
vector_store = AzureSearch(
    azure_search_endpoint=os.environ["AZURE_SEARCH_ENDPOINT"],
    credential=DefaultAzureCredential(),
    index_name="enterprise-knowledge-base",
    embedding_function=embeddings.embed_query
)

# Define the RAG prompt template
prompt_template = """Use the context below to answer the question.
If the answer is not in the context, say you do not have enough information.
Context: {context}
Question: {question}
Answer:"""

prompt = PromptTemplate(
    template=prompt_template,
    input_variables=["context", "question"]
)

# Assemble the RAG chain
rag_chain = RetrievalQA.from_chain_type(
    llm=llm,
    chain_type="stuff",
    retriever=vector_store.as_retriever(search_kwargs={"k": 5}),
    chain_type_kwargs={"prompt": prompt},
    return_source_documents=True
)

result = rag_chain({"query": "What is the refund policy for enterprise customers?"})
print(result["result"])
